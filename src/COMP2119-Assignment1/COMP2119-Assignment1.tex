        \begin{psol}{$\mathit{3n^{1.9}+n \notin \bt(n^2)}$}
            Consider the limit:
            $\di
                \lim_{n \to +\infty} \frac{3n^{1.9}+n}{n^2}

            $ which is not a non-zero constant.

            Thus, $\bm{3n^{1.9}+n \notin \bt(n^2)}$.
        \end{psol}

        \begin{psol}{$\mathit{2n^3 \notin \bom(10n^{3.1})}$}
            Assume for contradiction that $2n^3 \in \bom(10n^{3.1})$.
            Then, $\di \exists c, n_0 > 0 : \forall n \geq n_0, 2n^3 \geq 10cn^{3.1} \Rightarrow \frac{1}{n^{0.1}} \geq 5c \qquad [1]$.

            Pick $n = \max\left((5c)^{10}, n_0\right)$.
            Then, $\di \frac{1}{n^{0.1}} \leq \frac{1}{5c} < 5c$, which contradicts $[1]$.
            
            Thus, $\bm{2n^3 \notin \bom(10n^{3.1})}$.
        \end{psol}
